        /* Lọc sản phẩm theo danh mục & từ khóa phụ */
        function filterProducts(category = 'all', keyword = '') {
            // Cập nhật trạng thái nút Tab
            document.querySelectorAll('.filter-tab').forEach(btn => {
                const isActive = btn.dataset.cat === category;
                btn.className = isActive
                    ? 'filter-tab px-3 py-1.5 rounded-full text-xs font-semibold bg-brand text-white transition'
                    : 'filter-tab px-3 py-1.5 rounded-full text-xs font-semibold bg-white text-gray-700 border border-gray-300 hover:border-brand hover:text-brand transition';
            });

            let filtered = category === 'all'
                ? PRODUCTS
                : PRODUCTS.filter(p => p.category === category);

            const statusEl = document.getElementById('searchStatusMsg');
            if (keyword) {
                const kw = keyword.toLowerCase();
                const byKeyword = filtered.filter(p => p.name.toLowerCase().includes(kw) || p.specs.toLowerCase().includes(kw));
                if (byKeyword.length > 0) {
                    filtered = byKeyword;
                }
                statusEl.textContent = `Đang hiển thị mẫu sản phẩm liên quan: "${keyword}"`;
                statusEl.classList.remove('hidden');
            } else {
                statusEl.classList.add('hidden');
            }

            renderProducts(filtered);
        }

        /* Tìm kiếm trực tiếp trên thanh search Header */
        function performSearch(query) {
            const q = (query || '').trim().toLowerCase();
            const statusEl = document.getElementById('searchStatusMsg');
            const section = document.getElementById('san-pham-xingfa');
            if (section) section.scrollIntoView({ behavior: 'smooth' });

            if (!q) {
                filterProducts('all');
                return;
            }
            const results = PRODUCTS.filter(p =>
                p.name.toLowerCase().includes(q) ||
                p.categoryLabel.toLowerCase().includes(q) ||
                p.specs.toLowerCase().includes(q)
            );
            statusEl.textContent = `Kết quả tìm kiếm cho "${query}" (${results.length} sản phẩm)`;
            statusEl.classList.remove('hidden');
            renderProducts(results);
        }

        /* Logic 2 Banner Tự Trượt chuyên nghiệp + Hỗ trợ vuốt cảm ứng trên điện thoại */
        let currentBanner = 0;
        let bannerTimer = null;
        const bannerSlides = document.querySelectorAll('.hero-slide-item');
        const bannerDots = document.querySelectorAll('.banner-dot');

        function goToBanner(index) {
            currentBanner = (index + bannerSlides.length) % bannerSlides.length;
            bannerSlides.forEach((slide, idx) => {
                if (idx === currentBanner) {
                    slide.classList.add('is-active');
                    slide.classList.remove('is-hidden');
                } else {
                    slide.classList.remove('is-active');
                    slide.classList.add('is-hidden');
                }
            });
            bannerDots.forEach((dot, idx) => {
                if (idx === currentBanner) {
                    dot.classList.add('w-6', 'bg-amber-300');
                    dot.classList.remove('w-2.5', 'bg-white/50');
                } else {
                    dot.classList.remove('w-6', 'bg-amber-300');
                    dot.classList.add('w-2.5', 'bg-white/50');
                }
            });
            resetBannerAutoPlay();
        }

        function resetBannerAutoPlay() {
            if (bannerTimer) clearInterval(bannerTimer);
            bannerTimer = setInterval(() => {
                goToBanner(currentBanner + 1);
            }, 5000);
        }

        /* Điều khiển Menu Mobile */
        function openMobileMenu() {
            document.getElementById('mobileDrawerBackdrop').classList.remove('hidden');
            document.getElementById('mobileDrawer').classList.remove('-translate-x-full');
        }
        function closeMobileMenu() {
            document.getElementById('mobileDrawerBackdrop').classList.add('hidden');
            document.getElementById('mobileDrawer').classList.add('-translate-x-full');
        }
        function toggleMobileSub(id, btn) {
            const sub = document.getElementById(id);
            const icon = btn.querySelector('i');
            if (sub) sub.classList.toggle('hidden');
            if (icon) icon.classList.toggle('rotate-180');
        }

        document.addEventListener('DOMContentLoaded', () => {
            // Render toàn bộ sản phẩm mặc định
            renderProducts(PRODUCTS);

            // Bắt đầu tự trượt 2 banner
            resetBannerAutoPlay();
            document.getElementById('prevBannerBtn')?.addEventListener('click', () => goToBanner(currentBanner - 1));
            document.getElementById('nextBannerBtn')?.addEventListener('click', () => goToBanner(currentBanner + 1));

            // Hỗ trợ vuốt trái / phải chuyển banner trên điện thoại
            let touchStartX = 0;
            const sliderBox = document.getElementById('heroSliderContainer');
            if (sliderBox) {
                sliderBox.addEventListener('touchstart', (e) => {
                    touchStartX = e.changedTouches[0].screenX;
                }, { passive: true });
                sliderBox.addEventListener('touchend', (e) => {
                    const diff = e.changedTouches[0].screenX - touchStartX;
                    if (Math.abs(diff) > 45) {
                        goToBanner(diff < 0 ? currentBanner + 1 : currentBanner - 1);
                    }
                }, { passive: true });
            }

            // Sự kiện Menu Mobile & Tìm kiếm Mobile
            document.getElementById('mobileMenuBtn')?.addEventListener('click', openMobileMenu);
            document.getElementById('closeMobileMenuBtn')?.addEventListener('click', closeMobileMenu);
            document.getElementById('mobileDrawerBackdrop')?.addEventListener('click', closeMobileMenu);
            document.getElementById('mobileSearchToggle')?.addEventListener('click', () => {
                document.getElementById('mobileSearchBar')?.classList.toggle('hidden');
            });

            // Sự kiện Tìm kiếm Desktop & Mobile

            });
